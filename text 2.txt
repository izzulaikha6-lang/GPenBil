        dirLight.castShadow = true;

                star.style.top = startY + 'px';

                gsap.to(star, {
                    x: Math.cos(angle) * distance,
                    y: Math.sin(angle) * distance - 50,
                    rotation: Math.random() * 360,
                    opacity: 1,
                    duration: 0.5,
                    ease: "power2.out",
                    onComplete: () => {
                        gsap.to(star, {
                            opacity: 0,
                            duration: 0.5,
                            delay: 0.5,
                            onComplete: () => star.remove()
                        });
                    }
                });
            }
        }

        // Helper to convert base64 PCM to WAV for playback
        function base64ToArrayBuffer(base64) {
            const binaryString = window.atob(base64);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) {
                bytes[i] = binaryString.charCodeAt(i);
            }
            return bytes.buffer;
        }

        function pcmToWav(pcmData, sampleRate) {
            const numChannels = 1;
            const bitsPerSample = 16;
            const byteRate = sampleRate * numChannels * (bitsPerSample / 8);
            const blockAlign = numChannels * (bitsPerSample / 8);
            const dataSize = pcmData.length * 2;
            const buffer = new ArrayBuffer(44 + dataSize);
            const view = new DataView(buffer);

            const writeString = (view, offset, string) => {
                for (let i = 0; i < string.length; i++) {
                    view.setUint8(offset + i, string.charCodeAt(i));
                }
            };

            writeString(view, 0, 'RIFF');
            view.setUint32(4, 36 + dataSize, true);
            writeString(view, 8, 'WAVE');
            writeString(view, 12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true);
            view.setUint16(22, numChannels, true);
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, byteRate, true);
            view.setUint16(32, blockAlign, true);
            view.setUint16(34, bitsPerSample, true);
            writeString(view, 36, 'data');
            view.setUint32(40, dataSize, true);

            let offset = 44;
            for (let i = 0; i < pcmData.length; i++) {
                view.setInt16(offset, pcmData[i], true);
                offset += 2;
            }
            return new Blob([buffer], { type: 'audio/wav' });
        }

        async function generateText(userPrompt) {
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${API_KEY}`;
            
            // Add user message to history
            chatHistory.push({ role: "user", parts: [{ text: userPrompt }] });

            const payload = {
                contents: chatHistory,
                generationConfig: { temperature: 0.7 } // Slightly creative but focused
            };

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                const result = await response.json();
                if (result.candidates && result.candidates[0].content) {
                    const botReply = result.candidates[0].content.parts[0].text;
                    // Save bot reply to history
                    chatHistory.push({ role: "model", parts: [{ text: botReply }] });
                    return botReply;
                }
                throw new Error("Invalid format");
            } catch (error) {
                console.error("Text Gen Error:", error);
                // Remove the failed user prompt from history to keep it clean
                chatHistory.pop();
                return "Maaf, Kenyalang sedang pening sedikit. Boleh tanya semula?";
            }
        }

        async function generateSpeech(text) {
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${API_KEY}`;
            
            // Cleanup text for TTS (remove bold markdown, and strip Emojis that crash the TTS engine)
            const cleanText = text.replace(/\*\*/g, '').replace(/\*/g, '')
                                  .replace(/[\u{1F600}-\u{1F64F}\u{1F300}-\u{1F5FF}\u{1F680}-\u{1F6FF}\u{1F700}-\u{1F77F}\u{1F780}-\u{1F7FF}\u{1F800}-\u{1F8FF}\u{1F900}-\u{1F9FF}\u{1FA00}-\u{1FA6F}\u{1FA70}-\u{1FAFF}\u{2600}-\u{26FF}\u{2700}-\u{27BF}]/gu, '')
                                  .trim();

            const payload = {
                contents: [{ parts: [{ text: `Say cheerfully, clearly and speak quickly in Bahasa Melayu: ${cleanText}` }] }],
                generationConfig: {
                    responseModalities: ["AUDIO"],
                    speechConfig: {
                        voiceConfig: {
                            // Using Leda for a youthful, friendly tone suitable for a mascot
                            prebuiltVoiceConfig: { voiceName: "Leda" } 
                        }
                    }
                },
                model: "gemini-2.5-flash-preview-tts"
            };

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                const result = await response.json();
                
                const part = result?.candidates?.[0]?.content?.parts?.[0];
                if (part && part.inlineData && part.inlineData.data) {
                    const audioBase64 = part.inlineData.data;
                    const mimeType = part.inlineData.mimeType || "audio/pcm;rate=24000";
                    const sampleRateMatch = mimeType.match(/rate=(\d+)/);
                    const sampleRate = sampleRateMatch ? parseInt(sampleRateMatch[1], 10) : 24000;
                    
                    const pcmBuffer = base64ToArrayBuffer(audioBase64);
                    const pcm16 = new Int16Array(pcmBuffer);
                    const wavBlob = pcmToWav(pcm16, sampleRate);
                    return URL.createObjectURL(wavBlob);
                } else {
                    console.error("Invalid TTS Response:", result);
                }
            } catch (error) {
                console.error("TTS Error:", error);
            }
            return null;
        }

        async function handleSendMessage() {
            const message = chatInput.value.trim();
            if (!message) return;

            // Stop current audio if playing
            ttsAudio.pause();
            isSpeaking = false;

            // UI Updates
            chatInput.value = '';
            addMessageToUI('user', message);
            showTypingIndicator();

            // Check for gamification trigger words (client side evaluation logic enhancement)
            const lowerMsg = message.toLowerCase();
            const isPraiseExpected = lowerMsg.includes('betul') || lowerMsg.includes('hebat') || lowerMsg.includes('bagus');
            
            // 1. Get Text Response from Gemini
            let responseText = await generateText(isQuizMode ? `Saya pilih: ${message}. Adakah ini betul? Jika betul puji saya.` : message);
            
            removeTypingIndicator();
            lastBotMessage = responseText;

            // 2. Play animations based on text sentiment (simple heuristic)
            if (responseText.toLowerCase().includes('bagus') || responseText.toLowerCase().includes('tahniah') || responseText.toLowerCase().includes('hebat') || responseText.toLowerCase().includes('tepat')) {
                triggerHappyAnimation();
            } else {
                triggerNodAnimation();
            }

            // 3. Add to UI
            addMessageToUI('bot', responseText, true);

            // 4. Generate and Play Speech
            // Select the newly added play button to pass to playVoice
            const chatBubbles = chatHistoryEl.querySelectorAll('.chat-bot');
            const lastBubble = chatBubbles[chatBubbles.length - 1];
            const latestPlayBtn = lastBubble.querySelector('button');
            
            playVoice(responseText, latestPlayBtn);
        }

        async function playVoice(text, btnElement = null) {
             // Show speaking animation immediately while fetching audio
            isSpeaking = true;
            let originalHtml = "";
            
            if (btnElement) {
                originalHtml = btnElement.innerHTML;
                btnElement.innerHTML = '⏳ Memuat...';
                btnElement.disabled = true;
            }
            
            const audioUrl = await generateSpeech(text);
            
            if (audioUrl) {
                ttsAudio.src = audioUrl;
                ttsAudio.playbackRate = 1.15; // Diteguhkan pada paras selamat yang masih laju
                
                try {
                    await ttsAudio.play();
                } catch (e) {
                    console.error("Autoplay blocked or failed:", e);
                    isSpeaking = false;
                    if (btnElement) {
                        btnElement.innerHTML = '▶️ Sila tekan sekali lagi';
                        btnElement.disabled = false;
                    }
                    return; // Stop execution if play fails
                }
                
                ttsAudio.onended = () => {
                    isSpeaking = false;
                    if (btnElement) {
                        btnElement.innerHTML = originalHtml;
                        btnElement.disabled = false;
                    }
                };
                
                ttsAudio.onerror = () => {
                    isSpeaking = false;
                    console.error("Audio playback failed");
                    if (btnElement) {
                        btnElement.innerHTML = '❌ Gagal';
                        btnElement.disabled = false;
                    }
                }
            } else {
                isSpeaking = false;
                if (btnElement) {
                    btnElement.innerHTML = '❌ Ralat Sistem';
                    btnElement.disabled = false;
                }
            }
        }
